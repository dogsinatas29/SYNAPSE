# 1️⃣ SAFE SNAPSHOT REBUILD DESIGN (MANDATORY BASELINE)

## Core Flow

## Implementation

```ts
function filterSnapshot(snapshot: GraphSnapshot): GraphSnapshot {
  const ruleEngine = RuleEngine.getInstance()

  const nodes = snapshot.nodes.filter(n =>
    !ruleEngine.shouldIgnoreFile(n.filePath)
  )

  const nodeIds = new Set(nodes.map(n => n.id))

  const edges = snapshot.edges.filter(e =>
    nodeIds.has(e.from) && nodeIds.has(e.to)
  )

  const clusters = snapshot.clusters.filter(c =>
    c.nodeIds.some(id => nodeIds.has(id))
  )

  const cluster_flows = snapshot.cluster_flows?.filter(flow =>
    nodeIds.has(flow.from) && nodeIds.has(flow.to)
  ) ?? []

  return {
    ...snapshot,
    nodes,
    edges,
    clusters,
    cluster_flows
  }
}
```

## Execution

```ts
function rebuildWithBlacklist() {
  const snapshot = graphModel.createSnapshot()
  const filtered = filterSnapshot(snapshot)
  graphModel.restoreSnapshot(filtered)
}
```

# 2️⃣ AUTO SELECTION STRATEGY (PERFORMANCE CONTROL)

## Goal
Choose between:
- Full rebuild (safe, heavy)
- Incremental purge (fast, partial)

## Strategy

```ts
function shouldRebuild(totalNodes: number, affected: number): boolean {
  const ratio = affected / totalNodes

  if (totalNodes < 1000) return true
  if (ratio > 0.2) return true

  return false
}
```

## Execution

```ts
function applyBlacklistUpdate(affectedNodes: Node[]) {
  const total = graphModel.nodeCount()

  if (shouldRebuild(total, affectedNodes.length)) {
    rebuildWithBlacklist()
  } else {
    purgeIncremental(affectedNodes)
  }
}
```

```ts
function purgeIncremental(nodes: Node[], chunkSize = 100) {
  let index = 0

  function process() {
    const end = Math.min(index + chunkSize, nodes.length)

    for (; index < end; index++) {
      const node = nodes[index]
      if (isBlacklisted(node.path)) {
        graphModel.removeNode(node.id)
      }
    }

    if (index < nodes.length) {
      requestAnimationFrame(process)
    }
  }

  process()
}
```
