## 📌 Objective
Unified orchestration layer coordinating:
- GraphModel discard (real-time defense)
- Snapshot rebuild (consistency recovery)

Separate execution, unified control.

# 1️⃣ BlacklistOrchestrator Implementation

📄 src/core/BlacklistOrchestrator.ts

```ts
import { GraphModel } from './GraphModel'
import { filterSnapshot } from './filterSnapshot'
import { RuleEngine } from './RuleEngine'

export class BlacklistOrchestrator {
  private graphModel: GraphModel

  constructor(graphModel: GraphModel) {
    this.graphModel = graphModel
  }

  handleNodeAdd(node: any) {
    if (RuleEngine.getInstance().shouldIgnoreFile(node.filePath)) {
      return false
    }
    this.graphModel.addNode(node)
    return true
  }

  handleBlacklistChanged() {
    this.rebuild()
  }

  rebuild() {
    const snapshot = this.graphModel.createSnapshot()
    const filtered = filterSnapshot(snapshot)
    this.graphModel.restoreSnapshot(filtered)
  }
}
```
