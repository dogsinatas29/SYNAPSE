   ```typescript
       metricKey: 'controlScore' | 'amplificationScore'
   ): void {
       const idSet = new Set<string>();
       for (const c of components) {
           // 1. 입력 무결성 전수 검증 (§2.2 본문 명시)
           if (typeof c.id !== 'string' || c.id.trim() === '') {

           // 2단계: Component Path UTF-16 코드 단위 오름차순
   ```

### 2.4. 영속 저널 원자적 갱신 및 Crash-Durable 실행 규약

#### 1) 영속 저널 원자적 갱신 알고리즘 (`updateJournalAtomic`)
인플레이스 덮어쓰기로 인한 부분 기록/파손을 방지하기 위해 임시 파일 원자적 교체로 저널을 갱신합니다:

```typescript
function updateJournalAtomic(journalPath: string, data: TransactionJournal, dirFd: number): void {
    const tmpPath = journalPath + '.tmp';
    const fd = fs.openSync(tmpPath, fs.constants.O_CREAT | fs.constants.O_WRONLY | fs.constants.O_TRUNC);
    fs.writeFileSync(fd, JSON.stringify(data, null, 2), 'utf8');
    fs.fsyncSync(fd);
    fs.closeSync(fd);

    fs.renameSync(tmpPath, journalPath);
    fs.fsyncSync(dirFd); // 디렉터리 엔트리 영구 플러시
}
```

#### 2) Gate C2 Crash-Durable 실행 절차
1. **Step 1: 사전 검사, 버퍼 캡처 및 Inode 기록**:
   - 원본 SSoT `lstatSync` 검사 (`stat.isFile() && !stat.isSymbolicLink()`).
   - `sourceDev = stat.dev`, `sourceIno = stat.ino`, `sourceMtimeMs = stat.mtimeMs`.
   - 원본 바이트 스트림을 인메모리 버퍼(`origBuffer`)로 캡처하고 $H_{\text{orig}} = \text{sha256}(origBuffer)$ 계산.
2. **Step 2: 초기 저널 영속화 (`PREPARING`)**:
   - `updateJournalAtomic`을 통해 `state: 'PREPARING'` 저널 기록.
3. **Step 3: 내구성 백업 생성 및 저널 갱신 (`BACKED_UP`)**:
   - 캡처된 `origBuffer`를 `synapse_data/project_state.json.bak_c2`에 `O_CREAT | O_EXCL`로 기록 $\to$ `fsync(bak)` $\to$ `fsync(dir)`.
   - $H_{\text{bak}} = \text{sha256}(project\_state.json.bak_c2)$ 단언 ($H_{\text{orig}} \equiv H_{\text{bak}}$).
   - 저널 상태를 `BACKED_UP`으로 원자적 갱신 (`updateJournalAtomic`).
4. **Step 4: 임시 파일 기록 및 TOCTOU 방지 다중 속성 재검증**:
   - `synapse_data/project_state.json.tmp_c2`에 승인된 변환 결과 기록 (`O_CREAT | O_EXCL`) $\to$ `fsync(tmp)`.
   - **원본 Inode 및 mtimeMs 재검증**:
     `currStat = fs.lstatSync("synapse_data/project_state.json")`.
     단언: `currStat.dev === sourceDev && currStat.ino === sourceIno && currStat.mtimeMs === sourceMtimeMs`.
     (불일치 시 외부 프로세스에 의한 원본 교체/수정으로 간주, 변환 임시 파일 삭제 후 즉시 비상 정지).
5. **Step 5: 원자적 교체 및 저널 갱신 (`MUTATED`)**:
   - `fs.renameSync("synapse_data/project_state.json.tmp_c2", "synapse_data/project_state.json")`.
   - 디렉터리 `fsync(dirFd)`.
   - 저널 상태를 `MUTATED`로 원자적 갱신 (`updateJournalAtomic`).
6. **Step 6: 사후 검증 및 최종 커밋 (`COMMITTED`)**:
   - SSoT Diff 및 리포트 전수 검증 통과 시:
     저널 상태를 `COMMITTED`로 원자적 갱신 $\to$ 백업 파일, 저널 파일, 락 파일 안전 삭제.

### 2.5. 5대 Crash 복구 상태 판정 머신 및 롤백 알고리즘

#### 1) 내구성 롤백 실행 알고리즘 (`executeDurableRollback`)
```typescript
function executeDurableRollback(tx: TransactionJournal, dirFd: number): void {
    // 1. 백업 무결성 확인
    const bakHash = computeFileSha256(tx.backupPath);
    if (bakHash !== tx.hOrig) {
        throw new Error("EMERGENCY_MANUAL_LOCK: Backup file corrupted during rollback");
    }

    // 2. 복구 임시 파일 작성
    const restoreTmp = tx.sourcePath + '.restore_tmp';
    const fd = fs.openSync(restoreTmp, fs.constants.O_CREAT | fs.constants.O_WRONLY | fs.constants.O_TRUNC);
    const bakBuf = fs.readFileSync(tx.backupPath);
    fs.writeFileSync(fd, bakBuf);
    fs.fsyncSync(fd);
    fs.closeSync(fd);

    // 3. 원자적 복구 교체
    fs.renameSync(restoreTmp, tx.sourcePath);
    fs.fsyncSync(dirFd);

    // 4. 복구 결과 단언
    const restoredHash = computeFileSha256(tx.sourcePath);
    if (restoredHash !== tx.hOrig) {
        throw new Error("EMERGENCY_MANUAL_LOCK: Restored file hash mismatch");
    }

    // 5. 저널 상태 ROLLED_BACK 갱신
    tx.state = 'ROLLED_BACK';
    updateJournalAtomic(tx.sourcePath + '.journal', tx, dirFd);
}
```

#### 2) 5대 Crash 복구 상태 매트릭스
시스템 재부팅 시 저널 및 디스크 상태에 따라 다음 상태 전이를 강제합니다:

| 저널 기록 상태 | 실제 파일 상태 및 해시 대조 | 복구 동작 | 최종 결과 |
|---|---|---|:---:|
| **`PREPARING`** | 백업 생성 전 크래시 | 임시 파일/저널/락 파일 안전 삭제 | **CLEAN_ABORT** |
| **`BACKED_UP`** (Case 1) | $H_{\text{target}} \equiv H_{\text{orig}}$ (원본 불변) | 미적용 크래시이므로 임시/백업/저널/락 안전 삭제 | **CLEAN_ABORT** |
| **`BACKED_UP`** (Case 2) | $H_{\text{target}} \equiv H_{\text{candidate}}$ (교체 후 저널 갱신 전 크래시) | **불완전 트랜잭션으로 판정, `executeDurableRollback` 실행**하여 $H_{\text{orig}}$로 안전 복구 | **ROLLED_BACK $\to$ CLEAN_ABORT** |
| **`MUTATED`** | $H_{\text{target}} \equiv H_{\text{candidate}}$, $H_{\text{bak}} \equiv H_{\text{orig}}$ | Gate C1 리포트와 사후 검증 재수행. 통과 시 `COMMITTED`, 실패 시 `executeDurableRollback` | **COMMITTED / ROLLED_BACK** |
| **`ROLLED_BACK`** | $H_{\text{target}} \equiv H_{\text{orig}}$ 복구 확인 | 저널/백업/락 파일 안전 삭제 | **CLEAN_ABORT** |
| **비정상 상태** | 백업 파손($H_{\text{bak}} \not\equiv H_{\text{orig}}$), 원본 소실 또는 저널 파손 | **자동 복구 전면 금지**. 락/저널 유지한 채 시스템 비상 정지 | **`EMERGENCY_MANUAL_LOCK`** |

## 3. P0: 경로 정규화, UNC 대소문자 동등성 및 4단계 순차 파이프라인

### 3.1. UNC 대소문자 무관 동등성 판정 및 $W_{\text{canon}}$ 상대화 알고리즘

```typescript
export function getRelativePathIfInside(p: string, W_canon: string): string | null {
    // 1. POSIX 루트 특수 분기: W_canon === '/'
    if (W_canon === '/') {
        if (p === '/') return null; // 루트 자체는 파일이 아님
        if (p.startsWith('/') && !p.startsWith('//')) {
            return p.substring(1);
        }
        return null;
    }

    // 2. Windows 드라이브 루트 특수 분기: W_canon === 'C:/'
    if (/^[a-zA-Z]:\/$/.test(W_canon)) {
        const drive = W_canon.substring(0, 2).toUpperCase();
        if (/^[a-zA-Z]:\//.test(p) && p.substring(0, 2).toUpperCase() === drive) {
            if (p.toUpperCase() === W_canon.toUpperCase()) return null; // 드라이브 루트 자체
            return p.substring(3);
        }
        return null;
    }

    // 3. UNC 경로 특수 분기: W_canon === '//server/share'
    if (W_canon.startsWith('//')) {
        const uncPattern = /^\/\/([^/]+)\/([^/]+)(\/.*)?$/;
        const matchW = W_canon.match(uncPattern);
        const matchP = p.match(uncPattern);
        if (!matchW || !matchP) return null;

        // 서버명과 공유명은 대소문자 무관 비교
        if (matchW[1].toLowerCase() !== matchP[1].toLowerCase() ||
            matchW[2].toLowerCase() !== matchP[2].toLowerCase()) {
            return null;
        }

        const subW = (matchW[3] || '').replace(/^\/+/, '');
        const subP = (matchP[3] || '').replace(/^\/+/, '');

        // W_canon이 //server/share 루트 자체인 경우
        if (subW === '') {
            if (subP === '') return null; // UNC 루트 자체는 파일 노드가 아님 (null 반환)
            return subP;
        }

        // 하위 공유 폴더가 워크스페이스인 경우
        if (subW.toLowerCase() === subP.toLowerCase()) return null;
        const prefix = subW.endsWith('/') ? subW : subW + '/';
        if (subP.startsWith(prefix)) {
            return subP.substring(prefix.length);
        }
        return null;
    }

    // 4. 일반 Windows 드라이브 하위 경로: W_canon === 'C:/workspace'
    if (/^[a-zA-Z]:\//.test(W_canon)) {
        const drive = W_canon.substring(0, 2).toUpperCase();
        if (!/^[a-zA-Z]:\//.test(p) || p.substring(0, 2).toUpperCase() !== drive) {
            return null;
        }
        if (p.substring(0, 2).toUpperCase() + p.substring(2) === drive + W_canon.substring(2)) {
            return null; // 루트 자체
        }
        const pSub = drive + p.substring(2);
        const wSub = drive + W_canon.substring(2) + '/';
        if (pSub.startsWith(wSub)) {
            return pSub.substring(wSub.length);
        }
        return null;
    }

    // 5. 일반 POSIX 경로: W_canon === '/home/project'
    if (p === W_canon) return null;
    const prefix = W_canon.endsWith('/') ? W_canon : W_canon + '/';
    if (p.startsWith(prefix)) {
        return p.substring(prefix.length);
    }
    return null;
}
```

### 3.2. Step 0 가상 URI 선행 우회 및 4단계 순차 호출 파이프라인

가상 URI가 일반 파일시스템 슬래시 정규화나 `..` 검증에 의해 왜곡되지 않도록 **Step 0에서 선행 분리**합니다:

```
       ├─ (가상 URI: /^(ghost|external|builtin):\/\//i) ──> [Role 스키마 직행]

                                                                        │
┌───────────────────────────────────────────────────────────────────────┘
▼
[Step 3. validateNoWorkspaceEscape(p_rel)] ──> [Step 4. determineNodeRole(p_rel)]
```

1. **Step 0 (가상 URI 선행 우회)**:
   `/^(ghost|external|builtin):\/\//i.test(rawPath)` 성립 시, 파일 정규화 파이프라인을 완전 우회하고 가상 노드 스키마 Role(`GHOST`, `EXTERNAL`)로 즉시 직행.

### 4.2. 전역 적합성 모집단과 전이 인과관계 모집단 분리 및 판정 알고리즘

1. **모집단 1: 전역 Role 적합성 모집단 ($V_{\text{role-target}} \equiv V_{\text{NodeBuilder}}$)**:
   - 물리 파일 노드 전체에 대해 대상 구현이 완전 Oracle과 100% 일치함을 증명:
     $$\forall n \in V_{\text{role-target}}, \quad R_{\text{actual}}(n) \equiv R_{\text{full}}(n) \quad (\text{불일치 시 } \text{CONFORMANCE\_MISMATCH})$$
2. **모집단 2: 전이 인과관계 모집단 ($V_{\text{transition}}$)**:
   $$V_{\text{transition}} \triangleq \{ n \in V_{\text{role-target}} \mid R_{\text{orig}}(n) \neq \text{TEST} \land R_{\text{actual}}(n) \equiv \text{TEST} \}$$
3. **전이 인과 판정 알고리즘 (`classifyRoleTransition`)**:
   ```typescript
   export function classifyRoleTransition(
       r_orig: string,
       r_base: string,
       r_seg: string,
       r_file: string,
       r_full: string,
       r_actual: string
   ): 'PURE_SEGMENT_CAUSALITY' | 'PURE_FILE_CAUSALITY' | 'DUAL_CAUSALITY' | 'CONFORMANCE_MISMATCH' | 'NOT_A_TRANSITION_TARGET' | 'UNEXPLAINED_TRANSITION' {
       // 1. Role Enum 유효성 전수 검증
       assertValidRole(r_orig, 'r_orig');
       assertValidRole(r_base, 'r_base');
       assertValidRole(r_seg, 'r_seg');
       assertValidRole(r_file, 'r_file');
       assertValidRole(r_full, 'r_full');
       assertValidRole(r_actual, 'r_actual');

       // 2. 구현 적합성 검증
       if (r_actual !== r_full) {
           return 'CONFORMANCE_MISMATCH';
       }

       // 3. 최종 TEST 승격 불변식 검증 (r_full이 TEST가 아니면 승격 대상 아님)
       if (r_full !== 'TEST') {
           return 'NOT_A_TRANSITION_TARGET';
       }

       // 4. 원본 Role 불변식 검증 (기존에 이미 TEST였던 노드는 승격 인과 대상 아님)
       if (r_orig === 'TEST' || r_base === 'TEST') {
           return 'NOT_A_TRANSITION_TARGET';
       }

       // 5. 반사실적 인과관계 3대 분기
       const segIsTest = (r_seg === 'TEST');
       const fileIsTest = (r_file === 'TEST');

       if (segIsTest && fileIsTest) {
           return 'DUAL_CAUSALITY';
       }
       if (segIsTest && !fileIsTest) {
           return 'PURE_SEGMENT_CAUSALITY';
       }
       if (!segIsTest && fileIsTest) {
           return 'PURE_FILE_CAUSALITY';
       }

       return 'UNEXPLAINED_TRANSITION';
   }
   ```

```typescript
export function assertMappingDomainCompleteness(
    M_id: Map<string, string>,
    V_target: Set<string>,
    V_non_target: Set<string>

    // 1. 키 집합 양방향 일치 검증
    if (M_id.size !== V_target.size) {

            throw new Error("DOMAIN_KEY_MISSING: Target ID missing from M_id: " + targetId);
        }
    }
    for (const key of M_id.keys()) {

    }

    const valueSet = new Set<string>();
    for (const [key, val] of M_id.entries()) {

        if (valueSet.has(val)) {

        if (V_non_target.has(val)) {
            throw new Error("GLOBAL_DISJOINTNESS_VIOLATION: Mapped ID collides with non-target ID " + val);
        }
    }
}
```

### 5.2. `deepEqualJSON` 비교 계약 및 Edge 네거티브 필터링 불변식

```typescript
export function deepEqualJSON(a: any, b: any): boolean {
    if (a === b) return true;
    if (a === null || b === null || typeof a !== 'object' || typeof b !== 'object') {
        return false;
    }

    const aIsArray = Array.isArray(a);
    const bIsArray = Array.isArray(b);
    if (aIsArray !== bIsArray) return false;

    if (aIsArray) {
        if (a.length !== b.length) return false;
        for (let i = 0; i < a.length; i++) {
            if (!deepEqualJSON(a[i], b[i])) return false;
        }
        return true;
    }

    const keysA = Object.keys(a).filter(k => Object.prototype.hasOwnProperty.call(a, k)).sort();
    const keysB = Object.keys(b).filter(k => Object.prototype.hasOwnProperty.call(b, k)).sort();

    if (keysA.length !== keysB.length) return false;
    for (let i = 0; i < keysA.length; i++) {
        if (keysA[i] !== keysB[i]) return false;
        if (!deepEqualJSON(a[keysA[i]], b[keysB[i]])) return false;
    }

    return true;
}

// from, to만 제외하고 모든 잔여 메타데이터를 추출하는 네거티브 필터
export function getEdgeNonEndpointMetadata(e: Record<string, any>): Record<string, any> {
    const copy = { ...e };
    delete copy.from;
    delete copy.to;
    return copy;
}
```
