#### 2) 결정적 정규화 알고리즘 (Windows 드라이브 루트 완전 보존)
```typescript
export function auditNormalize(rawPath: string): string {
    // 1. Input Validation
    if (rawPath === null || rawPath === undefined) {
        throw new Error("NULL_OR_UNDEFINED_PATH_REJECTED");
    }
    if (typeof rawPath !== 'string') {
        throw new Error("INVALID_PATH_TYPE");
    }
    if (rawPath === '') return '';

    // 2. Virtual URI Bypass Check (대소문자 무관, 원본 100% 보존)
    if (/^(ghost|external|builtin):\/\//i.test(rawPath)) {
        return rawPath;
    }
    // file:// 스킴은 접두어 제거 후 로컬 경로로 변환
    let p = rawPath.replace(/^file:\/\//i, '');

    // 3. Separator Normalization
    p = p.replace(/\\/g, '/');

    // 4. Windows Drive Error & Relative Check
    // C: 또는 C:foo 와 같이 슬래시가 뒤따르지 않는 드라이브 상대 경로는 비정상 경로로 거부
    if (/^[a-zA-Z]:(?!\/)/.test(p)) {
        throw new Error("UNSUPPORTED_DRIVE_RELATIVE_PATH: " + rawPath);
    }

    // 5. Prefix Extraction & UNC / Drive / Absolute Detection
    let prefix = '';
    let isWindowsDrive = false;

    // Strict UNC: 정확히 2개의 선행 슬래시와 유효한 서버명/공유명 세그먼트
    if (/^\/\/[^\/\s]+\/[^\/\s]+/.test(p) && !p.startsWith('///')) {
        prefix = '//';
        p = p.substring(2);
    } else if (/^[a-zA-Z]:\//.test(p)) {
        // Windows Drive Root / Absolute Path: 접두사 C:/ 고정
        prefix = p.substring(0, 2) + '/';
        p = p.substring(2).replace(/^\/+/, '');
        isWindowsDrive = true;
    } else if (p.startsWith('/')) {
        prefix = '/';
        p = p.replace(/^\/+/, '');
    }

    // 6. Consecutive Slash Collapse
    p = p.replace(/\/+/g, '/');

    // 7. Segment Filtering & Parent Directory Preservation
    const parts = p.split('/');
    const resolvedParts: string[] = [];
    for (const part of parts) {
        if (part === '' || part === '.') {
            continue; // 빈 세그먼트 및 현재 디렉터리 축약
        }
        resolvedParts.push(part); // ".."를 포함한 모든 유의미 세그먼트 순서 보존
    }
    p = resolvedParts.join('/');

    // 8. Final Assembly & Trailing Handling
    if (prefix === '/' && p === '') return '/';
    if (prefix === '//' && p === '') return '/';
    if (isWindowsDrive && p === '') return prefix; // C:/ 루트 보존
    return prefix + p;
}
```

#### 3) 세그먼트 깊이 스택 기반 경로 이탈(Path Traversal) 검증 알고리즘

export function validateNoWorkspaceEscape(p_rel: string): void {
    if (p_rel === '' || /^(ghost|external|builtin):\/\//i.test(p_rel)) {
        return;
    }

    const segments = p_rel.split('/');
    let depth = 0;
    for (const seg of segments) {
        if (seg === '' || seg === '.') continue;
        if (seg === '..') {
            depth--;
            if (depth < 0) {
                throw new Error("WORKSPACE_ESCAPE_DETECTED: Path escapes workspace root via '..': " + p_rel);
            }
        } else {
            depth++;
        }
    }
}

### 3.2. $W_{\text{root}}$ Canonicalization, `isAbsolutePath` 및 외부 경로 정책 (`REJECT`)

#### 1) `isAbsolutePath(p: string): boolean` 완전 정의
```typescript
export function isAbsolutePath(p: string): boolean {
    if (!p || typeof p !== 'string') return false;
    // 가상 URI 스키마는 상대/절대 파일 경로가 아니므로 false
    if (/^(ghost|external|builtin):\/\//i.test(p)) return false;
    // POSIX 절대 경로 또는 UNC (선두가 슬래시로 시작)
    if (p.startsWith('/') || p.startsWith('\\')) return true;
    // Windows 드라이브 절대 경로 (C:/ 또는 C:\)
    if (/^[a-zA-Z]:[/\\]/.test(p)) return true;
    return false;
}
```

#### 2) $W_{\text{root}}$ Canonicalization 절차
워크스페이스 루트 경로는 비교 전 다음 표준화 절차를 거쳐 $W_{\text{canon}}$으로 고정됩니다:
1. $W_{\text{canon}} = \text{auditNormalize}(W_{\text{root}})$.
2. 루트 단독 경로(`/`, `C:/`)를 제외하고, 후행 슬래시가 존재할 경우 제거 (`if (W_canon.endsWith('/') && W_canon.length > 3) W_canon = W_canon.slice(0, -1)`).

#### 3) Lexical 경계 판정 규약
1. **가상 URI 스키마**: 스키마 기반 Role로 직행.
2. **상대 경로 (`!isAbsolutePath(p)`)**: $p_{\text{rel}} = p$. `validateNoWorkspaceEscape(p_rel)` 실행.
3. **루트 자체 ($p === W_{\text{canon}}$)**: 파일이 아닌 디렉터리 루트이므로 Role 분류 대상 제외 (`NOT_A_FILE_NODE`).
4. **내부 상대화**:
   - POSIX / UNC: `p.startsWith(W_canon + '/')`인 경우 $p_{\text{rel}} = p.\text{substring}(W_{\text{canon}}.\text{length} + 1)$.
   - Windows 드라이브: 드라이브 문자는 대소문자 무관 비교(`p.substring(0, 2).toUpperCase() === W_canon.substring(0, 2).toUpperCase()`), 하위 세그먼트는 lexical 접두사 대조(`p.substring(2).startsWith(W_canon.substring(2) + '/')`).
5. **외부 절대 경로 단일 정책: `REJECT`**:
   - 상기 조건을 만족하지 않는 모든 외부 절대 경로는 **즉시 거부 (`throw Error("EXTERNAL_ABSOLUTE_PATH_REJECTED")`)**.

### 3.4. `filePath` 1/2단계 검증의 완전 연결

1. **Stage 1: 입력 무결성 전수 검증 (Input Validation)**:
   - `typeof id !== 'string' || id.trim() === ""` $\to$ 에러 (`EMPTY_OR_NON_STRING_ID`).
   - `filePath === null` $\to$ 에러 (`NULL_FILE_PATH_REJECTED`).
   - `filePath !== undefined && typeof filePath !== 'string'` $\to$ 에러 (`INVALID_FILE_PATH_TYPE`).
   - `typeof filePath === 'string' && id !== filePath` $\to$ 에러 (`ID_FILE_PATH_MISMATCH`).
2. **Stage 2: 출력 불변식 검증 (Output Invariant Verification)**:
   변환 대상 노드 $u \in V_{\text{target}}$에 대해:
   - **원본에 `filePath`가 존재했던 경우**:
     $filePath' \equiv \text{auditNormalize}(filePath) \land id' \equiv \text{auditNormalize}(id) \land filePath' \equiv id'$.
   - **원본에 `filePath === undefined`였던 경우**:
     - **필수 생성 불변식**: 변환 노드 $u'$에 **반드시 `filePath` 필드가 생성되어야 함** (`u'.filePath !== undefined`).
     - **값 불변식**: $u'.filePath \equiv \text{auditNormalize}(u.id) \land u'.id \equiv \text{auditNormalize}(u.id)$.
   - **비대상 노드 ($u \in V \setminus V_{\text{target}}$)**:
     원본 객체 100% 보존 (`deepEqualJSON`).

## 5. P1: 전체 사상 $M^*$ 전역 단사성 및 Edge 실제 스키마 보존

### 5.1. $M^*(x)$의 수학적 정의, 정의역·공역 및 5단계 전역 단사성 증명

1. **정의역 및 공역 (Domain & Co-Domain)**:
   - $V$: 초기 SSoT의 전체 노드 ID 집합 (모든 ID는 고유한 유효 문자열).
   - $V_{\text{target}} \subseteq V$: Allow-list에 명시된 Phase B 변환 대상 노드 집합.
   - $M_{\text{id}}: V_{\text{target}} \to V'_{\text{target}}$: 대상 노드에 대한 1:1 전단사 ID 변환 함수 ($V'_{\text{target}} = \{ M_{\text{id}}(u) \mid u \in V_{\text{target}} \}$).
   - **전체 사상 함수 $M^*: V \to V'$**:
     $$M^*(x) \triangleq \begin{cases} 
     M_{\text{id}}(x), & \text{if } x \in V_{\text{target}} \\ 
     x, & \text{if } x \in V \setminus V_{\text{target}} 
     \end{cases}$$
   - 공역 $V' \triangleq M^*(V) = V'_{\text{target}} \cup (V \setminus V_{\text{target}})$.
2. **전역 단사성 증명 및 검증 5단계**:
   - **Step 1 (대상 내부 단사성)**: $\forall u_1, u_2 \in V_{\text{target}}, u_1 \neq u_2 \implies M_{\text{id}}(u_1) \neq M_{\text{id}}(u_2)$.
   - **Step 2 (전역 배타성, Global Disjointness)**:
     $$V'_{\text{target}} \cap (V \setminus V_{\text{target}}) = \varnothing$$
     변환된 대상 노드의 새 ID가 비대상 노드의 기존 ID와 겹치지 않음을 증명.
   - **Step 3 (비대상 항등 사상 단사성)**: $\forall v_1, v_2 \in V \setminus V_{\text{target}}, v_1 \neq v_2 \implies v_1 \neq v_2$.
   - **Step 4 (전역 단사성 도출)**: Step 1, 2, 3에 의해 $\forall x_1, x_2 \in V, x_1 \neq x_2 \implies M^*(x_1) \neq M^*(x_2)$.
   - **Step 5 (기수 불변 확인)**: $|V'| \equiv |V|$.

### 5.2. 실제 Edge 스키마 및 양방향 ID 일치

1. **SSoT 실제 Edge 스키마 확정**:
   SYNAPSE SSoT의 엣지는 정확히 다음 필드로 구성됩니다:
   ```typescript

       id: string;      // 고유 엣지 식별자 (필수)
       from: string;    // 출발 노드 ID (필수)
       to: string;      // 도착 노드 ID (필수)

   }
   ```
   *(끝점 필드명은 정확히 `from`, `to`이며 `source`/`target`이 아님을 확정)*
2. **Edge 보존 불변식**:
   - 원본 Edge ID 유일성 선행 조건: $\forall e_1 \neq e_2 \in E \implies e_1.\text{id} \neq e_2.\text{id}$.
