     ```typescript
     function compareCodeUnits(a: string, b: string): number {
         return a < b ? -1 : (a > b ? 1 : 0);
     }

     function sortComponentsDeterministically(components: ComponentFinding[], metricKey: 'controlScore' | 'amplificationScore'): void {
         // 선행 조건: 모든 항목의 ID 유일성 및 점수 유효성 검증
         const idSet = new Set<string>();
         for (const c of components) {
             if (typeof c.id !== 'string' || typeof c.componentPath !== 'string') {
                 throw new Error("INVALID_COMPONENT_DATA: id and componentPath must be strings");
             }
             if (idSet.has(c.id)) {
                 throw new Error("DUPLICATE_COMPONENT_ID_DETECTED: " + c.id);
             }
             idSet.add(c.id);
             const score = c[metricKey];
             if (typeof score !== 'number' || !Number.isFinite(score)) {
                 throw new Error("INVALID_SCORE_VALUE: NaN or non-finite detected on " + c.id);
             }
         }

         components.sort((a, b) => {
             // 1단계: 원본 메트릭 점수 내림차순
             const scoreA = a[metricKey];
             const scoreB = b[metricKey];
             if (scoreB !== scoreA) {
                 return scoreB - scoreA;
             }
             // 2단계: Component Path UTF-16 코드 단위 오름차순
             const pathOrder = compareCodeUnits(a.componentPath, b.componentPath);
             if (pathOrder !== 0) {
                 return pathOrder;
             }
             // 3단계: 고유 식별자 ID UTF-16 코드 단위 오름차순 (Tie-Breaker)
             return compareCodeUnits(a.id, b.id);
         });
     }
     ```

## 3. P0: 경로 정규화 알고리즘 및 $W_{\text{canon}}$ 경계 판정 규약

### 3.1. `auditNormalize` 입력 계약 및 Windows 드라이브 루트 보존 알고리즘

```typescript
export function auditNormalize(rawPath: string): string {
    // 1. Input Validation
    if (rawPath === null || rawPath === undefined) {
        throw new Error("NULL_OR_UNDEFINED_PATH_REJECTED");
    }
    if (typeof rawPath !== 'string') {
        throw new Error("INVALID_PATH_TYPE: Input must be a string");
    }
    if (rawPath === '') return '';

    // 2. Virtual URI Bypass Check (대소문자 무관, 원본 100% 보존)
    if (/^(ghost|external|builtin):\/\//i.test(rawPath)) {
        return rawPath;
    }
    // file:// 스킴은 접두어 제거 후 로컬 경로로 변환
    let p = rawPath.replace(/^file:\/\//i, '');

    // 3. Separator Normalization
    p = p.replace(/\\/g, '/');

    // 4. Windows Drive Error & Relative Check
    // C: 또는 C:foo 와 같이 슬래시가 뒤따르지 않는 드라이브 상대 경로는 비정상 경로로 거부
    if (/^[a-zA-Z]:(?!\/)/.test(p)) {
        throw new Error("UNSUPPORTED_DRIVE_RELATIVE_PATH: " + rawPath);
    }

    // 5. Prefix Extraction & UNC / Drive / Absolute Detection
    let prefix = '';
    let isWindowsDrive = false;

    // Strict UNC: 정확히 2개의 선행 슬래시와 유효한 서버명/공유명 세그먼트
    if (/^\/\/[^\/\s]+\/[^\/\s]+/.test(p) && !p.startsWith('///')) {
        prefix = '//';
        p = p.substring(2);
    } else if (/^[a-zA-Z]:\//.test(p)) {
        // Windows Drive Root / Absolute Path: 접두사 C:/ 고정
        prefix = p.substring(0, 2) + '/';
        p = p.substring(2).replace(/^\/+/, '');
        isWindowsDrive = true;
    } else if (p.startsWith('/')) {
        prefix = '/';
        p = p.replace(/^\/+/, '');
    }

    // 6. Consecutive Slash Collapse
    p = p.replace(/\/+/g, '/');

    // 7. Segment Filtering & Parent Directory Preservation
    const parts = p.split('/');
    const resolvedParts: string[] = [];
    for (const part of parts) {
        if (part === '' || part === '.') {
            continue; // 빈 세그먼트 및 현재 디렉터리 축약
        }
        resolvedParts.push(part); // ".."를 포함한 모든 유의미 세그먼트 순서 보존
    }
    p = resolvedParts.join('/');

    // 8. Final Assembly & Trailing Handling
    if (prefix === '/' && p === '') return '/';
    if (prefix === '//' && p === '') return '/';
    if (isWindowsDrive && p === '') return prefix; // C:/ 루트 보존
    return prefix + p;
}
```


#### 2) $W_{\text{canon}}$ Canonicalization 절차
1. $W_{\text{canon}} = \text{auditNormalize}(W_{\text{root}})$.
2. 루트 단독 경로(`/`, `C:/`)를 제외하고, 후행 슬래시가 존재할 경우 제거 (`if (W_canon.endsWith('/') && W_canon !== '/' && !/^[a-zA-Z]:\/$/.test(W_canon)) W_canon = W_canon.slice(0, -1)`).

#### 3) 루트 분기형 내부 상대화 알고리즘 (`getRelativePathIfInside`)
POSIX `/` 및 Windows `C:/` 루트에서 `//`가 생성되던 결함을 해결하는 공식 경계 판정 함수:

```typescript
export function getRelativePathIfInside(p: string, W_canon: string): string | null {
    if (p === W_canon) return null; // 워크스페이스 루트 자체는 파일이 아님

    // 1. POSIX 루트 특수 분기: W_canon === '/'
    if (W_canon === '/') {
        if (p.startsWith('/') && !p.startsWith('//')) {
            return p.substring(1); // 선행 슬래시만 제거 (예: '/home/a.ts' -> 'home/a.ts')
        }
        return null;
    }

    // 2. Windows 드라이브 루트 특수 분기: W_canon === 'C:/'
    if (/^[a-zA-Z]:\/$/.test(W_canon)) {
        const drive = W_canon.substring(0, 2).toUpperCase();
        if (/^[a-zA-Z]:\//.test(p) && p.substring(0, 2).toUpperCase() === drive) {
            return p.substring(3); // 'C:/' 접두사 제거 (예: 'C:/project/a.ts' -> 'project/a.ts')
        }
        return null;
    }

    // 3. 일반 Windows 드라이브 하위 경로: W_canon === 'C:/workspace'
    if (/^[a-zA-Z]:\//.test(W_canon)) {
        const drive = W_canon.substring(0, 2).toUpperCase();
        if (!/^[a-zA-Z]:\//.test(p) || p.substring(0, 2).toUpperCase() !== drive) {
            return null; // 다른 드라이브 또는 비-드라이브 경로
        }
        const pSub = drive + p.substring(2);
        const wSub = drive + W_canon.substring(2) + '/';
        if (pSub.startsWith(wSub)) {
            return pSub.substring(wSub.length);
        }
        return null;
    }

    // 4. 일반 POSIX / UNC 경로: W_canon === '/home/project' 또는 '//server/share'
    const prefix = W_canon.endsWith('/') ? W_canon : W_canon + '/';
    if (p.startsWith(prefix)) {
        return p.substring(prefix.length);
    }
    return null;
}
```

## 5. P1: 전체 사상 $M^*$ 전역 단사성 및 Edge 실제 스키마 보존

### 5.1. $M^*(x)$의 수학적 정의, 정의역·공역 및 5단계 전역 단사성 증명

1. **정의역 및 공역 (Domain & Co-Domain)**:
   - $V$: 초기 SSoT의 전체 노드 ID 집합 (모든 ID는 고유한 유효 문자열).
   - $V_{\text{target}} \subseteq V$: Allow-list에 명시된 Phase B 변환 대상 노드 집합.
   - $M_{\text{id}}: V_{\text{target}} \to V'_{\text{target}}$: 대상 노드 ID 변환 함수.
   - **전체 사상 함수 $M^*: V \to V'$**:
     $$M^*(x) \triangleq \begin{cases} 
     M_{\text{id}}(x), & \text{if } x \in V_{\text{target}} \\ 
     x, & \text{if } x \in V \setminus V_{\text{target}} 
     \end{cases}$$
   - 공역 $V' \triangleq M^*(V) = V'_{\text{target}} \cup (V \setminus V_{\text{target}})$.
2. **실제 변환 검증 5단계 절차**:
   - **Step 1 (대상 내부 단사성 검증)**: $\forall u \in V_{\text{target}}$, $M_{\text{id}}(u)$가 유효한 비어있지 않은 문자열이며, 생성된 Set 크기가 일치함을 확인 (`new Set(M_id.values).size === V_target.size`).
   - **Step 2 (전역 배타성 검증, Global Disjointness)**:
     $$V'_{\text{target}} \cap (V \setminus V_{\text{target}}) = \varnothing$$
     변환된 새 노드 ID가 비대상 노드의 기존 ID와 겹치지 않음을 검증.
   - **Step 3 (비대상 항등 사상 검증)**: $\forall v \in V \setminus V_{\text{target}}, M^*(v) \equiv v$.
   - **Step 4 (전역 단사성 도출)**: Step 1, 2, 3에 의해 $\forall x_1, x_2 \in V, x_1 \neq x_2 \implies M^*(x_1) \neq M^*(x_2)$.
   - **Step 5 (전체 기수 보존 검산)**: $|V'| \equiv |V|$.
